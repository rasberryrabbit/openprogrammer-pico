# Open Programmer firmware — Raspberry Pi Pico (RP2040) port
# Ported from the original PIC18 (MCC18) firmware v0.12.0 by Alberto Maccioni.
# Same USB HID wire protocol (VID 0x1209 / PID 0x5432) so the original
# host software (openprog GUI / CLI) works unchanged.
cmake_minimum_required(VERSION 3.13)

# Pull in the Pico SDK. pico_sdk_import.cmake must sit next to this file
# (the canonical copy shipped by pico-sdk at external/pico_sdk_import.cmake).
include(pico_sdk_import.cmake)

project(openprog_pico C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)

# Initialise the SDK; pio_stdio via USB is not used (we run a custom HID device).
pico_sdk_init()

# ---- PIO assembler (pioasm) integration ----------------------------------
# We use RP2040's PIO coprocessor for jitter-free bit-bang of One-Wire,
# JTAG, UNI/O and Microwire. pioasm compiles .pio source files into C
# headers at build time. The SDK ships pico_generate_pio_header() in
# tools/CMakeLists.txt; we include that here, and point it at a prebuilt
# pioasm binary.
#
# The pico_generate_pio_header() function looks for the executable in the
# `Pioasm` imported target (see pico-sdk/tools/CMakeLists.txt +
# FindPioasm.cmake). FindPioasm.cmake builds pioasm from source via
# ExternalProject, which fails under cmake 4.x (the bundled pioasm
# CMakeLists.txt requires cmake_minimum_required(VERSION 3.4)). We use a
# prebuilt pioasm binary, so we define the imported target directly and
# set Pioasm_FOUND to short-circuit FindPioasm.cmake.
#
# How to provide the pioasm binary (in priority order):
#   1. -DPICO_PIOASM_EXECUTABLE=/path/to/pioasm  on the cmake command line
#   2. $PICO_PIOASM_EXECUTABLE environment variable
#   3. ${PICO_SDK_PATH}/tools/pioasm/build/pioasm  (built in-tree)
#   4. `which pioasm`  (system-installed)
# If none is found, we fall back to FindPioasm.cmake's source build (works
# on cmake 3.x). See README.md "Building pioasm" for how to build pioasm
# from the pico-sdk sources.
if (NOT PICO_PIOASM_EXECUTABLE)
    if (DEFINED ENV{PICO_PIOASM_EXECUTABLE})
        set(PICO_PIOASM_EXECUTABLE $ENV{PICO_PIOASM_EXECUTABLE})
    elseif (EXISTS "${PICO_SDK_PATH}/tools/pioasm/build/pioasm")
        set(PICO_PIOASM_EXECUTABLE "${PICO_SDK_PATH}/tools/pioasm/build/pioasm")
    else()
        find_program(PICO_PIOASM_EXECUTABLE pioasm)
    endif()
endif()
if (PICO_PIOASM_EXECUTABLE AND NOT TARGET Pioasm)
    add_executable(Pioasm IMPORTED)
    set_property(TARGET Pioasm PROPERTY IMPORTED_LOCATION ${PICO_PIOASM_EXECUTABLE})
    set(Pioasm_FOUND 1)
    message(STATUS "Using prebuilt pioasm: ${PICO_PIOASM_EXECUTABLE}")
endif()
include(${PICO_SDK_PATH}/tools/CMakeLists.txt)

# ---- Application sources -------------------------------------------------
set(OPENPROG_SOURCES
    src/main.c
    src/usb_descriptors.c
    src/usb_app.c
    src/hal.c
    src/opcontrol.c
    src/opcontrol_helpers.c
    src/pio_drivers.c
)

add_executable(${CMAKE_PROJECT_NAME} ${OPENPROG_SOURCES})

target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

# Compile the .pio programs into autogenerated C headers (ow.pio.h,
# unio.pio.h) under the build directory. The headers are included by
# pio_drivers.c. JTAG and Microwire are host-driven SIO bit-bang, so
# jtag.pio / uw.pio are no longer compiled (their SMs were never used and
# pio_gpio_init() would have stolen the bit-bang pins' FUNCSEL).
pico_generate_pio_header(${CMAKE_PROJECT_NAME} ${CMAKE_CURRENT_SOURCE_DIR}/src/ow.pio)
pico_generate_pio_header(${CMAKE_PROJECT_NAME} ${CMAKE_CURRENT_SOURCE_DIR}/src/unio.pio)

# TinyUSB provides the HID class; the pico-sdk pulls it in via the
# pico_tinyusb library. We enable the device stack + HID class.
target_link_libraries(${CMAKE_PROJECT_NAME}
    pico_stdlib
    pico_bootsel_via_double_reset
    hardware_adc
    hardware_pwm
    hardware_dma
    hardware_irq
    hardware_flash
    hardware_pio
    tinyusb_device
    tinyusb_board
)

# USB identifiers (match the original PIC firmware so the existing host
# software enumerates the device without changes).
target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE
    # TinyUSB descriptors are provided by us (usb_descriptors.c).
    # Board-specific USB identifiers are NOT used; we override via tusb_config.
    PICO_XOSC_STARTUP_DELAY_MS=64
)

# Generate standard Pico output artefacts: .uf2 (drag-flash), .hex, .elf, .map.
pico_add_extra_outputs(${CMAKE_PROJECT_NAME})

# Helpful: print size + map file alongside the .uf2.
pico_set_program_name(${CMAKE_PROJECT_NAME} "openprog")
pico_set_program_version(${CMAKE_PROJECT_NAME} "0.12.0-pico")

# Disable default USB stdio (we run a real HID device on the same USB port).
pico_enable_stdio_usb(${CMAKE_PROJECT_NAME} 0)
pico_enable_stdio_uart(${CMAKE_PROJECT_NAME} 0)
